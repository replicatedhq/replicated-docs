```yaml
apiVersion: v1
kind: Secret
metadata:
  name: prometheus-support-bundle
  labels:
    troubleshoot.sh/kind: support-bundle
stringData:
  support-bundle-spec: |-
    apiVersion: troubleshoot.sh/v1beta2
    kind: SupportBundle
    metadata:
      name: prometheus
    spec:
      collectors:
        - runPod:
            collectorName: prometheus-up
            name: prometheus-up
            namespace: my-app
            timeout: 60s
            podSpec:
              containers:
                - name: curl
                  image: curlimages/curl:8.16.0
                  command: ["curl"]
                  args:
                    - "-sS"
                    - "--max-time"
                    - "10"
                    - "http://prometheus-server.monitoring.svc/api/v1/query?query=up"
      analyzers:
        - textAnalyze:
            checkName: Prometheus query API
            fileName: prometheus-up/prometheus-up.log
            regex: '"status":"success"'
            outcomes:
              - pass:
                  when: "true"
                  message: The Prometheus query API responded successfully.
              - fail:
                  when: "false"
                  message: The Prometheus query did not succeed. Check `prometheus-up/prometheus-up.log` in the bundle for the curl or Prometheus error.
```
